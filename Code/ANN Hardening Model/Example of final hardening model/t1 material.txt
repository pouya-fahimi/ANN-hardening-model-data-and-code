      subroutine HARDENING(inp,f6)

      implicit none

      integer HLSIZE
      REAL*8 MAXA(1,14),MINA(1,14)
      double precision, DIMENSION(30,13) :: w1
      double precision, DIMENSION(30,1) :: b1
      double precision, DIMENSION(30,30) :: w2
      double precision, DIMENSION(30,1) :: b2
      double precision, DIMENSION(30,30) :: w3
      double precision, DIMENSION(30,1) :: b3
      double precision, DIMENSION(30,30) :: w4
      double precision, DIMENSION(30,1) :: b4
      double precision, DIMENSION(30,30) :: w5
      double precision, DIMENSION(30,1) :: b5
      double precision, DIMENSION(1,30) :: w6
      double precision, DIMENSION(1,1) :: b6
      double precision :: f6(1,1)
      double precision :: inp(1,13)


      call readnetwork (w1,b1,w2,b2,w3,b3,w4,b4,w5,b5,w6,b6,HLSIZE,MINA,MAXA)

      call neet (inp,MINA,MAXA,w1,b1,w2,b2,w3,b3,w4,b4,w5,b5,w6,b6,f6)

      end subroutine HARDENING



      subroutine neet (inp,MINA,MAXA,w1,b1,w2,b2,w3,b3,w4,b4,w5,b5,w6,b6,f6)

      implicit none

      double precision :: inp(1,13),Ninp(1,13),Ninpt(13,1)
      REAL*8 MAXA(1,14),MINA(1,14),y
      double precision :: w1(30,13),w2(30,30),w3(30,30),w4(30,30),w5(30,30),w6(1,30) 
      double precision :: b1(30,1),b2(30,1),b3(30,1),b4(30,1),b5(30,1),b6(1,1) 
      double precision, allocatable :: f1(:,:),f2(:,:),f3(:,:),f4(:,:),f5(:,:) 
      double precision :: f6(1,1)
      integer I

      DO I=1,13
            Ninp(1,I)=(inp(1,I)-MINA(1,I))/( MAXA(1,I)-MINA(1,I))
      ENDDO

      Ninpt = transpose(Ninp)

      f1=(matmul(w1,Ninpt))+b1
      DO I=1,size(b1, 1)
            call tansig (f1(I,1),y)
            f1(I,1)=y
      ENDDO

      f2=(matmul(w2,f1))+b2
      DO I=1,size(b2, 1)
            call tansig (f2(I,1),y)
            f2(I,1)=y
      ENDDO

      f3=(matmul(w3,f2))+b3
      DO I=1,size(b3, 1)
            call tansig (f3(I,1),y)
            f3(I,1)=y
      ENDDO

      f4=(matmul(w4,f3))+b4
      DO I=1,size(b4, 1)
            call tansig (f4(I,1),y)
            f4(I,1)=y
      ENDDO

      f5=(matmul(w5,f4))+b5
      DO I=1,size(b5, 1)
            call tansig (f5(I,1),y)
            f5(I,1)=y
      ENDDO

      f6=(matmul(w6,f5))+b6
      f6=f6*(MAXA(1,14)-MINA(1,14))+MINA(1,14)

      end subroutine neet
